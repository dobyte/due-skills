# 网格服务器 (Mesh)

## 概述

网格服务器（mesh）即微服务，是 Due 框架集群架构中的无状态业务处理单元。支持 GRPC 和 RPCX 两种主流RPC框架，用于构建分布式微服务架构。

## 包路径

```go
import "github.com/dobyte/due/v2/cluster/mesh"
```

## 配置参数

```toml
[cluster.mesh]
    # 实例ID，集群中唯一。不填写默认自动生成唯一的实例ID
    id = ""
    # 实例名称
    name = "mesh"
    # 编解码器。可选：json | proto。默认为proto
    codec = "proto"
    # 服务权重，用于服务发现的加权负载均衡，权重值必需大于0才生效。默认为1
    weight = 1
    # 实例元数据
    [cluster.mesh.metadata]
        # 键值对，且均为字符串类型
        key = "value"
    # 集群内RPC连接器配置
    [cluster.mesh.linker]
        # RPC连接数，默认5
        connNum = 5
        # RPC调用超时时间，默认为3s
        callTimeout = "3s"
        # RPC拨号超时时间，默认3s
        dialTimeout = "3s"
        # RPC拨号重试次数，默认3
        dialRetryTimes = 3
        # RPC连接故障恢复时间，默认5s
        faultRecoveryTime = "5s"
        # 命令队列大小，默认4096
        commandQueueSize = 4096
        # 命令写入超时时间，默认0s（不超时）
        commandWriteTimeout = "0s"

[transport.grpc.server]
    # 服务器监听地址。空或:0时系统将会随机端口号
    addr = ":0"
    # 是否将内部通信地址暴露到公网。默认为false
    expose = false
    # 私钥文件
    keyFile = ""
    # 证书文件
    certFile = ""

[transport.grpc.client]
    # CA证书文件
    caFile = ""
    # 证书域名
    serverName = ""
```

网格服务器至少需要注册一个服务提供者（`AddServiceProvider`），否则启动时会因 `no service registered` 而终止。

### 网格代理

```go
type Proxy struct {
    // 实例信息与状态
    GetID() string
    GetName() string
    GetState() cluster.State
    SetState(state cluster.State) error

    // 服务提供者与钩子
    AddServiceProvider(name string, desc, provider any)
    AddHookListener(hook cluster.Hook, handler HookHandler)

    // 服务调用
    NewMeshClient(target string) (transport.Client, error)

    // 网关/节点链接与用户定位
    HasGate(gid string) (bool, error)
    AskGate(ctx context.Context, gid string, uid int64) (string, bool, error)
    LocateGate(ctx context.Context, uid int64) (string, error)
    BindGate(ctx context.Context, gid string, cid, uid int64) error
    UnbindGate(ctx context.Context, uid int64) error
    FetchGateList(ctx context.Context, states ...cluster.State) ([]*registry.ServiceInstance, error)
    HasNode(nid string) (bool, error)
    AskNode(ctx context.Context, uid int64, name, nid string) (string, bool, error)
    LocateNode(ctx context.Context, uid int64, name string) (string, error)
    LocateNodes(ctx context.Context, uid int64) (map[string]string, error)
    BindNode(ctx context.Context, uid int64, name, nid string) error
    UnbindNode(ctx context.Context, uid int64, name, nid string) error
    FetchNodeList(ctx context.Context, states ...cluster.State) ([]*registry.ServiceInstance, error)

    // 消息处理与会话推送
    Deliver(ctx context.Context, args *cluster.DeliverArgs) error
    Push(ctx context.Context, args *cluster.PushArgs) error
    Multicast(ctx context.Context, args *cluster.MulticastArgs) (int64, error)
    Broadcast(ctx context.Context, args *cluster.BroadcastArgs) (int64, error)
    Publish(ctx context.Context, args *cluster.PublishArgs) (int64, error)
    Subscribe(ctx context.Context, args *cluster.SubscribeArgs) error
    Unsubscribe(ctx context.Context, args *cluster.UnsubscribeArgs) error
    GetIP(ctx context.Context, args *cluster.GetIPArgs) (string, error)
    Stat(ctx context.Context, kind session.Kind) (int64, error)
    IsOnline(ctx context.Context, args *cluster.IsOnlineArgs) (bool, error)
    Disconnect(ctx context.Context, args *cluster.DisconnectArgs) error

    // 消息打包
    PackMessage(message *cluster.Message) ([]byte, error)
    PackBuffer(message any) ([]byte, error)
}
```

## GRPC 示例

### 服务端

```go
package main

import (
    "context"
    "github.com/dobyte/due/registry/consul/v2"
    "github.com/dobyte/due/transport/grpc/v2"
    "github.com/dobyte/due/v2"
    "github.com/dobyte/due/v2/cluster/mesh"
)

type Server struct {
    pb.UnimplementedGreeterServer
    proxy *mesh.Proxy
}

func NewServer(proxy *mesh.Proxy) *Server {
    return &Server{proxy: proxy}
}

func (s *Server) Init() {
    s.proxy.AddServiceProvider("greeter", &pb.Greeter_ServiceDesc, s)
}

func (s *Server) Hello(ctx context.Context, args *pb.HelloArgs) (*pb.HelloReply, error) {
    return &pb.HelloReply{Message: "Hello " + args.Name}, nil
}

func main() {
    // 创建容器
    container := due.NewContainer()
    // 创建服务发现
    registry := consul.NewRegistry()
    // 创建RPC传输器
    transporter := grpc.NewTransporter()
    // 创建网格组件
    component := mesh.NewMesh(
        mesh.WithRegistry(registry),
        mesh.WithTransporter(transporter),
    )
    // 初始化应用
    initAPP(component.Proxy())
    // 添加网格组件
    container.Add(component)
    // 启动容器
    container.Serve()
}

func initAPP(proxy *mesh.Proxy) {
    server.NewServer(proxy).Init()
}
```

### 客户端

```go
package main

import (
    "context"

    "github.com/dobyte/due/registry/consul/v2"
    duegrpc "github.com/dobyte/due/transport/grpc/v2"
    "github.com/dobyte/due/v2/log"
    "google.golang.org/grpc"
)

func main() {
    // 创建服务发现
    registry := consul.NewRegistry()
    // 构建传输器
    transporter := duegrpc.NewTransporter()
    // 设置默认的服务发现组件
    transporter.SetDefaultDiscovery(registry)

    // 构建客户端
    client, err := transporter.NewClient("discovery://greeter")
    if err != nil {
        log.Errorf("create rpc client failed: %v", err)
        return
    }

    // 获取底层 gRPC 连接
    conn := client.Client().(grpc.ClientConnInterface)

    // 创建服务客户端
    cli := pb.NewGreeterClient(conn)

    // 发起RPC调用
    reply, err := cli.Hello(context.Background(), &pb.HelloArgs{Name: "fuxiao"})
    if err != nil {
        log.Errorf("invoke rpc func failed: %v", err)
        return
    }

    log.Infof("invoke rpc func replay: %v", reply)
}
```

## RPCX 示例

### 服务端

```go
package main

import (
    "context"
    "github.com/dobyte/due/registry/consul/v2"
    "github.com/dobyte/due/transport/rpcx/v2"
    "github.com/dobyte/due/v2"
    "github.com/dobyte/due/v2/cluster/mesh"
)

const (
    service     = "greeter"
    servicePath = "Greeter"
)

type Server struct {
    proxy *mesh.Proxy
}

func NewServer(proxy *mesh.Proxy) *Server {
    return &Server{proxy: proxy}
}

func (s *Server) Init() {
    s.proxy.AddServiceProvider(service, servicePath, s)
}

func (s *Server) Hello(ctx context.Context, args *pb.HelloArgs, reply *pb.HelloReply) error {
    reply.Message = "Hello " + args.Name
    return nil
}

func main() {
    container := due.NewContainer()
    registry := consul.NewRegistry()
    transporter := rpcx.NewTransporter()
    
    component := mesh.NewMesh(
        mesh.WithRegistry(registry),
        mesh.WithTransporter(transporter),
    )
    
    initAPP(component.Proxy())
    container.Add(component)
    container.Serve()
}

func initAPP(proxy *mesh.Proxy) {
    server.NewServer(proxy).Init()
}
```

### 客户端

```go
package main

import (
    "github.com/dobyte/due/registry/consul/v2"
    "github.com/dobyte/due/transport/rpcx/v2"
    "github.com/dobyte/due/v2/log"
    rpcxclient "github.com/smallnest/rpcx/client"
)

func main() {
    // 创建服务发现
    registry := consul.NewRegistry()
    // 构建传输器
    transporter := rpcx.NewTransporter()
    // 设置默认的服务发现组件
    transporter.SetDefaultDiscovery(registry)

    // 构建客户端
    client, err := transporter.NewClient("discovery://greeter")
    if err != nil {
        log.Errorf("create rpc client failed: %v", err)
        return
    }

    // 获取底层 rpcx 客户端（*rpcxclient.OneClient）
    rpcxClient := client.Client().(*rpcxclient.OneClient)

    // 可基于 rpcxClient 构建代码生成的服务客户端并发起调用
    _ = rpcxClient
}
```

## 最佳实践

1. **服务设计**:
   - 遵循单一职责原则
   - 合理划分服务边界

2. **协议选择**:
   - GRPC 适合高性能场景
   - RPCX 适合需要更多特性的场景

3. **服务发现**:
   - 使用 `discovery://` 前缀进行服务发现
   - 配合注册中心实现动态路由

4. **错误处理**:
   - 实现重试机制提高可用性
   - 设置合理的超时时间

5. **监控告警**:
   - 监控RPC调用延迟和成功率
   - 记录关键业务指标
